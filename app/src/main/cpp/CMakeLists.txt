cmake_minimum_required(VERSION 3.22.1)
project(mangajanai LANGUAGES CXX)
add_library(ncnn SHARED IMPORTED)
set_target_properties(ncnn PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ncnn/lib/${ANDROID_ABI}/libncnn.so")
add_library(mangajanai SHARED mangajanai.cpp)
target_include_directories(mangajanai PRIVATE vendor/ncnn/include)
target_compile_features(mangajanai PRIVATE cxx_std_17)
target_compile_options(mangajanai PRIVATE -O3 -fexceptions -frtti)
target_link_options(mangajanai PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(mangajanai PRIVATE ncnn log android)
